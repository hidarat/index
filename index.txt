        function registerSector() {
            let sName = document.getElementById('regSectorName').value.trim();

            let aPass = document.getElementById('regAdminPass').value.trim();
            let email = document.getElementById('regEmail').value.trim();
            let phone = document.getElementById('regPhone').value.trim();

            if (!sName || !aPass || !aName) return;

            let code = 'SEC-' + Math.floor(1000 + Math.random() * 9000);

            sectors[code] = { 
                name: sName, adminName: aName, pass: aPass, email: email, phone: phone,
                bgUrl: '', bgOpacity: '0.15', headerLogoUrl: '', appTitle: "نظام إدارة وإحصاء الإنجازات",
                isFree: false, // يبدأ برسوم
                hasPaid: false, // لم يتم دفع الرسوم بعد (لا يظهر العداد)
                isDisabled: false, 
                startDate: new Date().toISOString().split('T')[0], 
                expiryDate: '',
                departments: [], tasks: [], employees: [], achievements: [], limitedAdmins: [] 
            };

            localStorage.setItem('sectors_db', JSON.stringify(sectors));
            alert(`تم إنشاء الحساب بنجاح!\nكود القطاع: [ ${code} ]\nملاحظة: الحساب يتطلب دفع رسوم الاشتراك وتحديد المدة لفتح كافة الموارد وإظهار العداد.`);
            showPage('adminLogin');
            document.getElementById('loginAdminSectorCode').value = code;
        }

        function checkSubscriptionExpiry(code) {
            let sec = sectors[code];
            if (sec.isDisabled) { alert('هذا القطاع موقوف.'); return false; }
            if (sec.isFree) return true; // مجاني بالكامل
            
            // إذا لم يدفع الرسوم بعد
            if (!sec.hasPaid) {
                return true; // يُسمح بالدخول لصفحة التحكم لإظهار تنبيه أو زر الدفع وتحديد المدة
            }

            let today = new Date().toISOString().split('T')[0];
            if (sec.expiryDate && sec.expiryDate < today) { 
                alert('عذراً، انتهت صلاحية اشتراك هذا الحساب! يرجى سداد رسوم الاشتراك وتحديد المدة لتجديده.'); 
                return false; 
            }
            return true;
        }

        function empLogin() {
            let code = document.getElementById('empSectorCode').value.trim();
            let user = document.getElementById('empUsername').value.trim();
            let pass = document.getElementById('empPassword').value.trim();
            let dept = document.getElementById('empDeptSelect').value;

            if(!sectors[code]) { alert('كود القطاع غير صحيح'); return; }
            if(!checkSubscriptionExpiry(code)) return;
            if(!user || !pass) { alert('أدخل اسم الموظف والرقم السري'); return; }

            let sec = sectors[code];
            let existingEmp = sec.employees.find(e => (typeof e === 'object' ? e.name : e) === user);

            let isFullAccess = sec.isFree || sec.hasPaid;

            if(existingEmp) {
                if(typeof existingEmp === 'object' && existingEmp.pass && existingEmp.pass !== pass) {
                    alert('الرقم السري غير صحيح'); return;
                }
            } else {
                if(!isFullAccess && sec.employees.length >= 4) {
                    currentSectorCode = code;
                    document.getElementById('upgradeReasonText').textContent = "لقد وصلت للحد الأقصى للموظفين (4 موظفين فقط). يرجى سداد رسوم الاشتراك وتحديد المدة لفتح كافة الموارد.";
                    document.getElementById('upgradeModal').classList.remove('hidden');
                    return;
                }
                sec.employees.push({ name: user, pass: pass, dept: dept });
                localStorage.setItem('sectors_db', JSON.stringify(sectors));
            }

            currentSectorCode = code;
            currentEmpUser = user;
            currentUserRole = 'employee';
            applyBackgroundStyle(sec.bgUrl || '', sec.bgOpacity || '0.15');
            applyHeaderLogo(sec.headerLogoUrl || '');
            applyAppTitleText(sec.appTitle || "نظام إدارة وإحصاء الإنجازات");

            let taskSelect = document.getElementById('empTaskSelect');
            taskSelect.innerHTML = '<option value="">-- اختر اسم المهمة --</option>';
            sec.tasks.forEach(t => { taskSelect.innerHTML += `<option value="${t}">${t}</option>`; });

            document.getElementById('empSectorBadge').innerHTML = `القطاع: <b>${sec.name}</b> | الإدارة: <b>${dept}</b> | الموظف: <b>${user}</b>`;
            updateEmpDashboardStats();

            closeUpgradeModal();
            loadAdminDashboard();
        }

        function updateEmpDashboardStats() {
            let counts = {};
            sectors[currentSectorCode].achievements.forEach(a => { counts[a.empUser] = (counts[a.empUser] || 0) + 1; });
            let sortedEmps = Object.keys(counts).map(emp => ({ name: emp, count: counts[emp] })).sort((a, b) => b.count - a.count);
            renderHonorBoardAndTopTen(sortedEmps, 'empHonorBoardContainer', 'empTopEmployeeName', 'empTopEmployeeStats', 'empTopTenTable');
        }

        document.getElementById('empSectorCode').addEventListener('input', function() {
            let code = this.value.trim();
            let deptSelect = document.getElementById('empDeptSelect');
            deptSelect.innerHTML = '<option value="عام">-- الإدارة العامة --</option>';
            if(sectors[code] && sectors[code].departments) {
                sectors[code].departments.forEach(d => { deptSelect.innerHTML += `<option value="${d}">${d}</option>`; });
            }
        });

        function renderHonorBoardAndTopTen(sortedEmps, boardId, nameId, statsId, tableId) {
            let board = document.getElementById(boardId);
            if(sortedEmps.length > 0) {
                board.classList.remove('hidden');
                document.getElementById(nameId).textContent = `⭐ ${sortedEmps[0].name} ⭐`;
                document.getElementById(statsId).textContent = `بإجمالي ${sortedEmps[0].count} مهمة`;
            } else { board.classList.add('hidden'); }

            let topTenTable = document.getElementById(tableId);
            topTenTable.innerHTML = '';
            sortedEmps.slice(0, 10).forEach((item, index) => {
                topTenTable.innerHTML += `<tr><td>المركز ${index + 1}</td><td>${item.name}</td><td><b>${item.count}</b> مهمة</td></tr>`;
            });
        }

        function saveAchievement() {
            let taskName = document.getElementById('empTaskSelect').value;
            let taskNum = document.getElementById('empTaskNumber').value.trim();
            let taskContent = document.getElementById('empTaskContent').value.trim();
            let date = document.getElementById('empAchieveDate').value;

            if(!taskName || !taskNum || !taskContent || !date) return;
            sectors[currentSectorCode].achievements.push({ empUser: currentEmpUser, taskName, taskNum, taskContent, date });
            localStorage.setItem('sectors_db', JSON.stringify(sectors));
            alert('تم حفظ الإنجاز بنجاح');
            document.getElementById('empTaskNumber').value = '';
            document.getElementById('empTaskContent').value = '';
            loadEmpHistory();
            updateEmpDashboardStats();
        }

        function isPastMonth(dateString) {
            if (!dateString) return false;
            let d = new Date(dateString), now = new Date();
            return (d.getFullYear() < now.getFullYear() || (d.getFullYear() === now.getFullYear() && d.getMonth() < now.getMonth()));
        }

        function loadEmpHistory() {
            let table = document.getElementById('empHistoryTable');
            table.innerHTML = '';
            let allAch = sectors[currentSectorCode].achievements;
            let empAchIndices = [];
            allAch.forEach((a, i) => { if(a.empUser === currentEmpUser) empAchIndices.push(i); });

            if(empAchIndices.length === 0) { table.innerHTML = '<tr><td colspan="5" style="text-align:center;">لا توجد إنجازات</td></tr>'; return; }

            empAchIndices.forEach(idx => {
                let a = allAch[idx];
                let isLocked = isPastMonth(a.date);
                let actions = isLocked ? '<span class="locked-tag">🔒 مقفل</span>' : `<button class="btn btn-warning" onclick="openEditModal(${idx})">تعديل</button><button class="btn btn-danger" onclick="deleteEmpAchievement(${idx})">حذف</button>`;
                table.innerHTML += `<tr><td>${a.date}</td><td><b>${a.taskName}</b></td><td>${a.taskNum}</td><td>${a.taskContent}</td><td>${actions}</td></tr>`;
            });
        }

        function deleteEmpAchievement(idx) {
            if (isPastMonth(sectors[currentSectorCode].achievements[idx].date)) return;
            if (confirm('حذف الإنجاز؟')) {
                sectors[currentSectorCode].achievements.splice(idx, 1);
                localStorage.setItem('sectors_db', JSON.stringify(sectors));
                loadEmpHistory();
                updateEmpDashboardStats();
            }
        }

        function openEditModal(idx) {
            editingAchIndex = idx;
            let ach = sectors[currentSectorCode].achievements[idx];
            if (isPastMonth(ach.date)) return;
            let select = document.getElementById('editTaskSelect');
            select.innerHTML = '';
            sectors[currentSectorCode].tasks.forEach(t => {
                select.innerHTML += `<option value="${t}" ${t===ach.taskName?'selected':''}>${t}</option>`;
            });
            document.getElementById('editTaskNumber').value = ach.taskNum;
            document.getElementById('editTaskContent').value = ach.taskContent;
            document.getElementById('editAchieveDate').value = ach.date;
            document.getElementById('editAchModal').classList.remove('hidden');
        }

        function closeEditModal() { document.getElementById('editAchModal').classList.add('hidden'); }

        function saveEditedAchievement() {
            if (editingAchIndex > -1) {
                let ach = sectors[currentSectorCode].achievements[editingAchIndex];
                ach.taskName = document.getElementById('editTaskSelect').value;
                ach.taskNum = document.getElementById('editTaskNumber').value.trim();
                ach.taskContent = document.getElementById('editTaskContent').value.trim();
                ach.date = document.getElementById('editAchieveDate').value;
                localStorage.setItem('sectors_db', JSON.stringify(sectors));
                closeEditModal();
                loadEmpHistory();
                updateEmpDashboardStats();
            }
        }

        function renderChart() {
            let data = sectors[currentSectorCode];
            let counts = {};
            data.achievements.forEach(a => { counts[a.empUser] = (counts[a.empUser] || 0) + 1; });
            let labels = Object.keys(counts), values = Object.values(counts);

            let ctx = document.getElementById('adminChart').getContext('2d');
            if(window.myAdminChart) window.myAdminChart.destroy();
            window.myAdminChart = new Chart(ctx, {
                data: {
                    labels: labels.length > 0 ? labels : ['لا بيانات'],
                    datasets: [
                        { type: 'line', label: 'نقاط', data: values.length > 0 ? values : [0], borderColor: '#e53e3e', fill: false },
                        { type: 'bar', label: 'مهام', data: values.length > 0 ? values : [0], backgroundColor: 'rgba(43, 108, 176, 0.75)' }
                    ]
                },
                options: { responsive: true, maintainAspectRatio: false, scales: { y: { beginAtZero: true, ticks: { stepSize: 1 } } } }
            });
        }
